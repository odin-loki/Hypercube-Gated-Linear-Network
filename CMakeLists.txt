# HGLN with FORGE (https://github.com/odin-loki/FORGE) for SIMD dispatch and GPU execution.
#   cmake -S . -B build/forge -G Ninja -DCMAKE_CXX_COMPILER=clang++-18 -DCMAKE_C_COMPILER=clang-18
#   cmake --build build/forge && ctest --test-dir build/forge
# The plain Makefile still builds the dependency-free header-only library and its tests.
cmake_minimum_required(VERSION 3.25)
project(HGLN VERSION 5.0.0 LANGUAGES C CXX)

option(HGLN_WITH_FORGE "Build the FORGE SIMD-dispatch and GPU backends" ON)
option(HGLN_BUILD_TESTS "Build HGLN tests" ON)
set(HGLN_FORGE_SOURCE_DIR "" CACHE PATH "Local FORGE checkout (copied and patched in the build tree); empty = fetch FORGE_GIT_TAG")
set(FORGE_GIT_REPOSITORY "https://github.com/odin-loki/FORGE" CACHE STRING "FORGE repository")
set(FORGE_GIT_TAG "b135941680664502fec59924f51d3c3dc11e23e9" CACHE STRING "FORGE commit the patches apply to")

if(HGLN_BUILD_TESTS)
  enable_testing()
endif()
find_package(Python3 COMPONENTS Interpreter)

# Header-only core library (C++17; also built standalone by the Makefile).
add_library(hgln INTERFACE)
target_include_directories(hgln INTERFACE ${PROJECT_SOURCE_DIR}/include)
find_package(Threads REQUIRED)
target_link_libraries(hgln INTERFACE Threads::Threads)

if(HGLN_WITH_FORGE)
  include(cmake/hgln_forge.cmake)   # obtains, patches and adds FORGE (targets forge_simd, forge_simd_mv, forge_gpu, ...)
  include(cmake/hgln_simd.cmake)    # hgln_simd: per-target CPU kernels chosen at run time through forge::simd
  include(cmake/hgln_gpu.cmake)     # hgln_gpu: the HGLN kernel run through forge::gpu::cuda_hip (CUDA and HIP)
endif()

# CLI and tests shared by both backends
add_executable(hgln_cli src/hgln_cli.cpp)
target_link_libraries(hgln_cli PRIVATE hgln $<$<BOOL:${HGLN_WITH_FORGE}>:hgln_simd>)
target_compile_definitions(hgln_cli PRIVATE $<$<BOOL:${HGLN_WITH_FORGE}>:HGLN_WITH_FORGE=1>)
add_executable(hgln_cli_f src/hgln_cli.cpp)
target_link_libraries(hgln_cli_f PRIVATE hgln $<$<BOOL:${HGLN_WITH_FORGE}>:hgln_simd>)
target_compile_definitions(hgln_cli_f PRIVATE USE_FLOAT $<$<BOOL:${HGLN_WITH_FORGE}>:HGLN_WITH_FORGE=1>)
# version 5.0 integer net (include/hgln_q.hpp): hgln_q bench | classify | parity (vs research/alg's reference)
add_executable(hgln_cli_q src/hgln_cli.cpp)
set_target_properties(hgln_cli_q PROPERTIES OUTPUT_NAME hgln_q)
target_link_libraries(hgln_cli_q PRIVATE hgln $<$<BOOL:${HGLN_WITH_FORGE}>:hgln_simd>)
target_compile_definitions(hgln_cli_q PRIVATE USE_INT $<$<BOOL:${HGLN_WITH_FORGE}>:HGLN_WITH_FORGE=1>)
if(HGLN_BUILD_TESTS)
  add_executable(hgln_test_api tests/test_api.cpp)
  target_link_libraries(hgln_test_api PRIVATE hgln $<$<BOOL:${HGLN_WITH_FORGE}>:hgln_simd>)
  target_compile_definitions(hgln_test_api PRIVATE $<$<BOOL:${HGLN_WITH_FORGE}>:HGLN_WITH_FORGE=1>)
  add_test(NAME hgln_api COMMAND hgln_test_api)
  add_executable(hgln_test_q tests/test_q.cpp)
  target_link_libraries(hgln_test_q PRIVATE hgln $<$<BOOL:${HGLN_WITH_FORGE}>:hgln_simd>)
  target_compile_definitions(hgln_test_q PRIVATE $<$<BOOL:${HGLN_WITH_FORGE}>:HGLN_WITH_FORGE=1>)
  add_test(NAME hgln_q_reference COMMAND hgln_test_q)
  foreach(u 8 12 16)
    foreach(b 16 12)
      add_test(NAME hgln_q_parity_u${u}_b${b}_d8 COMMAND hgln_cli_q parity 8 4500 ${u} ${b} 3)
      add_test(NAME hgln_q_parity_u${u}_b${b}_d64 COMMAND hgln_cli_q parity 64 1500 ${u} ${b} 4)
    endforeach()
  endforeach()
  if(Python3_FOUND)
    foreach(u 8 12 16)
      foreach(m v43 v42)
        foreach(d 8 64)
          add_test(NAME hgln_parity_${m}_u${u}_f64_d${d} COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/python/parity.py $<TARGET_FILE:hgln_cli> ${d} 1e-12 ${m} ${u})
          add_test(NAME hgln_parity_${m}_u${u}_f32_d${d} COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/python/parity.py $<TARGET_FILE:hgln_cli_f> ${d} 1e-4 ${m} ${u})
        endforeach()
      endforeach()
    endforeach()
    foreach(d 8 64)   # fan-out: layer 3 reads layer 1 (Params::fanout)
      add_test(NAME hgln_parity_v43f_u16_f64_d${d} COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/python/parity.py $<TARGET_FILE:hgln_cli> ${d} 1e-12 v43f 16)
      add_test(NAME hgln_parity_v43f_u16_f32_d${d} COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/python/parity.py $<TARGET_FILE:hgln_cli_f> ${d} 1e-4 v43f 16)
    endforeach()
    # version 5.0 options: cross-layer latch, latch credit, gate error, top-K corners. Top-K modes (v5, v5c, v43k) in
    # float64 only: float32 rounding resolves near-ties of the hard selection differently (python/parity.py)
    foreach(u 8 12 16)
      foreach(m v5 v5d v5c)
        foreach(d 8 64)
          add_test(NAME hgln_parity_${m}_u${u}_f64_d${d} COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/python/parity.py $<TARGET_FILE:hgln_cli> ${d} 1e-12 ${m} ${u})
        endforeach()
      endforeach()
      foreach(d 8 64)
        add_test(NAME hgln_parity_v5d_u${u}_f32_d${d} COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/python/parity.py $<TARGET_FILE:hgln_cli_f> ${d} 1e-4 v5d ${u})
        add_test(NAME hgln_parity_v43k_u${u}_f64_d${d} COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/python/parity.py $<TARGET_FILE:hgln_cli> ${d} 1e-12 v43k ${u})
      endforeach()
    endforeach()
    add_test(NAME hgln_digits COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/python/digits_test.py $<TARGET_FILE:hgln_cli> 3)
    add_test(NAME hgln_q_digits COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/python/digits_test.py $<TARGET_FILE:hgln_cli_q> 3)
  endif()
endif()
